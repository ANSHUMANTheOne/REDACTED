// --- UPDATED CREATE MATCHER WITH ERROR HANDLING --- async function createMatcher() { const status = document.getElementById('scan-status'); const descriptors = []; for (const label of authorized) { try { status.innerText = `ENCODING: ${label.toUpperCase()}...`; // Try to fetch the image const img = await faceapi.fetchImage(`./${label}.png`); // Detect face in the photo const det = await faceapi.detectSingleFace(img) .withFaceLandmarks() .withFaceDescriptor(); if (det) { descriptors.push(new faceapi.LabeledFaceDescriptors(label, [det.descriptor])); console.log(`✅ ${label} encoded successfully.`); } else { console.warn(`⚠️ No face found in ${label}.png. Skipping...`); // This prevents the "stuck" screen if the photo is blurry } } catch (e) { console.error(`❌ Failed to load ${label}.png:`, e); status.innerText = `ERROR_LOADING: ${label.toUpperCase()}`; } } if (descriptors.length === 0) { status.innerText = "WARNING: NO_AUTHORIZED_FACES_LOADED"; // Create an empty matcher so the rest of the app doesn't break return new faceapi.FaceMatcher([], 0.6); } return new faceapi.FaceMatcher(descriptors, 0.6); } // --- UPDATED MODEL LOADING (More Reliable URL) --- async function initGuestOS() { initChart(); const status = document.getElementById('scan-status'); status.innerText = "LOADING_AI_MODELS..."; // We append a trailing slash to ensure the library finds the manifest files const MODEL_URL = 'https://cdn.jsdelivr.net/npm/@vladmandic/face-api/model/'; try { await Promise.all([ faceapi.nets.ssdMobilenetv1.loadFromUri(MODEL_URL), faceapi.nets.faceLandmark68Net.loadFromUri(MODEL_URL), faceapi.nets.faceRecognitionNet.loadFromUri(MODEL_URL), faceapi.nets.faceExpressionNet.loadFromUri(MODEL_URL) ]); faceMatcher = await createMatcher(); status.innerText = "GATEWAY_ONLINE"; startCamera(); } catch (err) { status.innerText = "FATAL_ERROR: MODEL_LOAD_FAILED"; console.error("AI Model Error:", err); } }
