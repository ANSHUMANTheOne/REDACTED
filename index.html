async function createMatcher() { const status = document.getElementById('scan-status'); const descriptors = []; for (const label of authorized) { try { status.innerText = `ENCODING: ${label.toUpperCase()}...`; // 1. Force fetch with no-cache to ensure we get the latest file const img = await faceapi.fetchImage(`./${label}.png`); // 2. Use TinyFaceDetector if SsdMobilenet fails (it's faster for PNGs) const det = await faceapi.detectSingleFace(img).withFaceLandmarks().withFaceDescriptor(); if (det) { descriptors.push(new faceapi.LabeledFaceDescriptors(label, [det.descriptor])); console.log(`✅ ${label} encoded.`); } else { // IMPORTANT: This prevents the 'stuck' screen! console.warn(`⚠️ Face not detected in ${label}.png - Check lighting in photo.`); } } catch (e) { console.error(`❌ Could not load ${label}.png. Check if file exists in folder.`); } } // 3. Final Check: Don't let the app crash if no faces are loaded if (descriptors.length === 0) { status.innerText = "GATEWAY_READY (GUEST_MODE_ONLY)"; return new faceapi.FaceMatcher([], 0.6); } status.innerText = "GATEWAY_READY (AUTH_MODE)"; return new faceapi.FaceMatcher(descriptors, 0.6); }
